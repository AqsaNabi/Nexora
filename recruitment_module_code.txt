import os
import json
from pathlib import Path

from dotenv import load_dotenv
from google_auth_oauthlib.flow import Flow

# Project root: Nexora/
BASE_DIR = Path(__file__).resolve().parents[2]

# Load environment variables from Nexora/.env
load_dotenv(BASE_DIR / ".env")

TOKEN_FILE = Path(__file__).resolve().parent / "token.json"

CLIENT_ID = os.getenv("GOOGLE_CLIENT_ID")
CLIENT_SECRET = os.getenv("GOOGLE_CLIENT_SECRET")
PROJECT_ID = os.getenv("GOOGLE_PROJECT_ID")
REDIRECT_URI = os.getenv("GOOGLE_REDIRECT_URI")

SCOPES = [
    "https://www.googleapis.com/auth/forms.body",
    "https://www.googleapis.com/auth/forms.responses.readonly",
    "https://www.googleapis.com/auth/drive.file"
]


def get_client_config():
    if not all([CLIENT_ID, CLIENT_SECRET, PROJECT_ID, REDIRECT_URI]):
        raise ValueError(
            "Google OAuth environment variables are missing. "
            "Please check your .env file."
        )

    return {
        "web": {
            "client_id": CLIENT_ID,
            "project_id": PROJECT_ID,
            "client_secret": CLIENT_SECRET,
            "auth_uri": "https://accounts.google.com/o/oauth2/auth",
            "token_uri": "https://oauth2.googleapis.com/token",
            "redirect_uris": [REDIRECT_URI]
        }
    }


def create_oauth_flow():
    return Flow.from_client_config(
        get_client_config(),
        scopes=SCOPES,
        redirect_uri=REDIRECT_URI,
        autogenerate_code_verifier=True
    )


def get_authorization_url():
    flow = create_oauth_flow()

    authorization_url, state = flow.authorization_url(
        access_type="offline",
        include_granted_scopes="true",
        prompt="consent"
    )

    return authorization_url, state, flow
def exchange_code_for_token(code, flow):
    flow.fetch_token(code=code)

    credentials = flow.credentials

    token_data = {
        "token": credentials.token,
        "refresh_token": credentials.refresh_token,
        "token_uri": credentials.token_uri,
        "client_id": credentials.client_id,
        "client_secret": credentials.client_secret,
        "scopes": credentials.scopes
    }

    with open(TOKEN_FILE, "w", encoding="utf-8") as token_file:
        json.dump(token_data, token_file, indent=4)

    return credentials

from database.recruitment_db import get_recruitment_connection


def create_recruitment_tables():
    connection = None
    cursor = None

    try:
        connection = get_recruitment_connection()
        cursor = connection.cursor()

        # 1. Recruitment posts
        cursor.execute("""
            CREATE SEQUENCE IF NOT EXISTS recruitment_post_id_seq
            START WITH 1 INCREMENT BY 1;
        """)

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS recruitment_posts (
                post_id VARCHAR(20) PRIMARY KEY
                    DEFAULT (
                        'POST-' ||
                        LPAD(
                            NEXTVAL('recruitment_post_id_seq')::TEXT,
                            6,
                            '0'
                        )
                    ),
                job_description TEXT NOT NULL
            );
        """)

        # 2. Universal applicant data
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS recruitment_data (
                application_id BIGSERIAL PRIMARY KEY,
                post_id VARCHAR(20) NOT NULL,
                full_name VARCHAR(150) NOT NULL,
                email VARCHAR(150) NOT NULL,
                phone VARCHAR(20),
                resume_file BYTEA,
                resume_filename VARCHAR(255),
                resume_content_type VARCHAR(100),
                additional_data JSONB DEFAULT '{}'::jsonb,
                submitted_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

                CONSTRAINT fk_recruitment_post
                    FOREIGN KEY (post_id)
                    REFERENCES recruitment_posts(post_id)
                    ON DELETE CASCADE
            );
        """)

        # 3. Recruitment metadata
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS recruitment_metadata (
                metadata_id BIGSERIAL PRIMARY KEY,
                post_id VARCHAR(20) NOT NULL UNIQUE,
                form_id VARCHAR(255),
                form_url TEXT,
                image_urls JSONB DEFAULT '[]'::jsonb,
                created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
                updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

                CONSTRAINT fk_metadata_post
                    FOREIGN KEY (post_id)
                    REFERENCES recruitment_posts(post_id)
                    ON DELETE CASCADE
            );
        """)

        connection.commit()

        print("All three recruitment tables created successfully!")

    except Exception as error:
        if connection:
            connection.rollback()

        print(f"Table creation failed: {error}")
        raise

    finally:
        if cursor:
            cursor.close()

        if connection:
            connection.close()


if __name__ == "__main__":
    create_recruitment_tables()

import json

from backend.database.recruitment_db import get_recruitment_connection


def create_recruitment_post(job_description):
    """
    Creates a recruitment post and returns its generated post ID.
    """

    connection = None
    cursor = None

    try:
        connection = get_recruitment_connection()
        cursor = connection.cursor()

        cursor.execute(
            """
            INSERT INTO recruitment_posts (job_description)
            VALUES (%s)
            RETURNING post_id;
            """,
            (job_description,)
        )

        post_id = cursor.fetchone()[0]
        connection.commit()

        return post_id

    except Exception as error:
        if connection:
            connection.rollback()
        raise Exception(f"Failed to create recruitment post: {error}")

    finally:
        if cursor:
            cursor.close()
        if connection:
            connection.close()



def save_recruitment_metadata(
    post_id,
    form_id=None,
    form_url=None,
    image_urls=None,
    question_mapping=None
):
    """
    Saves Google Form details, generated image URLs,
    and Google question ID to field_name mappings.
    """

    connection = None
    cursor = None

    try:
        connection = get_recruitment_connection()
        cursor = connection.cursor()

        image_urls = image_urls or []
        question_mapping = question_mapping or {}

        cursor.execute(
            """
            INSERT INTO recruitment_metadata (
                post_id,
                form_id,
                form_url,
                image_urls,
                question_mapping
            )
            VALUES (%s, %s, %s, %s::jsonb, %s::jsonb)
            RETURNING metadata_id;
            """,
            (
                post_id,
                form_id,
                form_url,
                json.dumps(image_urls),
                json.dumps(question_mapping)
            )
        )

        metadata_id = cursor.fetchone()[0]
        connection.commit()

        return metadata_id

    except Exception as error:
        if connection:
            connection.rollback()
        raise Exception(f"Failed to save recruitment metadata: {error}")

    finally:
        if cursor:
            cursor.close()
        if connection:
            connection.close()



def get_recruitment_post(post_id):
    """
    Retrieves a recruitment post with its metadata.
    """

    connection = None
    cursor = None

    try:
        connection = get_recruitment_connection()
        cursor = connection.cursor()

        cursor.execute(
            """
            SELECT
                p.post_id,
                p.job_description,
                m.form_id,
                m.form_url,
                m.image_urls,
                m.created_at
            FROM recruitment_posts p
            LEFT JOIN recruitment_metadata m
                ON p.post_id = m.post_id
            WHERE p.post_id = %s;
            """,
            (post_id,)
        )

        result = cursor.fetchone()

        if not result:
            return None

        return {
            "post_id": result[0],
            "job_description": result[1],
            "form_id": result[2],
            "form_url": result[3],
            "image_urls": result[4],
            "created_at": result[5]
        }

    except Exception as error:
        raise Exception(f"Failed to retrieve recruitment post: {error}")

    finally:
        if cursor:
            cursor.close()
        if connection:
            connection.close()


def get_applicants_by_post(post_id):
    """
    Retrieves all applicants associated with a recruitment post.
    Does not return the actual PDF bytes.
    """

    cursor = None

    try:
        connection = get_recruitment_connection()
        cursor = connection.cursor()

            """
            SELECT
                application_id,
                post_id,
                full_name,
                email,
                phone,
                resume_filename,
                resume_content_type,

                submitted_at

                "full_name": row[2],
                "email": row[3],
                "phone": row[4],
                "resume_filename": row[5],
                "resume_content_type": row[6],
                "additional_data": row[7],
                "submitted_at": row[8]
            }
            for row in rows
        ]

    except Exception as error:
        raise Exception(f"Failed to retrieve applicants: {error}")

    finally:
        if cursor:
            cursor.close()
        if connection:
            connection.close()

from fastapi import APIRouter, WebSocket, WebSocketDisconnect
from starlette.concurrency import run_in_threadpool

from backend.services.chat_service import ChatService

from backend.services.approved_payload_service import (
    create_approved_payload
)

from backend.services.google_form_service import (
    create_google_form_from_json
)

from backend.services.google_watch_service import (
    create_response_watch
)

from backend.services.job_post_service import (
    prepare_job_post,
    prepare_application_form
)

from backend.repository.recruitment_repository import (
    create_recruitment_post,
    save_recruitment_metadata
)


router = APIRouter()
active_sessions = {}


@router.websocket("/ws/chat/{session_id}")
async def chat_websocket(websocket: WebSocket, session_id: str):

    await websocket.accept()

    if session_id not in active_sessions:
        active_sessions[session_id] = ChatService()

    chat_service = active_sessions[session_id]

    await websocket.send_json({
        "type": "connected",
        "message": "WebSocket connected successfully."
    })

    async def create_form_if_approved():

        if not (
            chat_service.job_post_approved
            and chat_service.application_form_approved
        ):
            return

        if getattr(chat_service, "recruitment_metadata_saved", False):
            return

        if getattr(chat_service, "form_creation_in_progress", False):
            return

        chat_service.form_creation_in_progress = True

        try:
            # 1. Prepare approved JSON in memory.
            payload = create_approved_payload(
                session_id,
                chat_service.current_job_post,
                chat_service.current_application_form
            )

            job_post = payload["job_post"]
            application_form = payload["application_form"]

            job_description = job_post["job_description"]

            # 2. Save recruitment post in PostgreSQL.
            if not getattr(chat_service, "created_post_id", None):

                chat_service.created_post_id = await run_in_threadpool(
                    create_recruitment_post,
                    job_description
                )

            # 3. Create Google Form directly from approved JSON.
            if not getattr(chat_service, "created_form_id", None):

                result = await run_in_threadpool(
                    create_google_form_from_json,
                    payload
                )

                chat_service.created_form_id = result["form_id"]
                chat_service.created_form_url = result["form_url"]
                chat_service.question_mapping = result["question_mapping"]

            # 4. Save Google Form metadata in PostgreSQL.
            if not getattr(chat_service, "recruitment_metadata_saved", False):

                await run_in_threadpool(
                    save_recruitment_metadata,
                    chat_service.created_post_id,
                    chat_service.created_form_id,
                    chat_service.created_form_url,
                    None,
                    chat_service.question_mapping
                )

                chat_service.recruitment_metadata_saved = True

            # 5. Register Google Forms response notifications.
            if not getattr(chat_service, "response_watch_created", False):

                await run_in_threadpool(
                    create_response_watch,
                    chat_service.created_form_id
                )

                chat_service.response_watch_created = True

            # 6. Notify frontend.
            await websocket.send_json({
                "type": "form_created",
                "message": "Recruitment post and Google Form created successfully.",
                "session_id": payload["session_id"],
                "post_id": chat_service.created_post_id,
                "form_id": chat_service.created_form_id,
                "form_url": chat_service.created_form_url
            })

        except Exception as error:

            await websocket.send_json({
                "type": "error",
                "message": str(error)
            })

        finally:
            chat_service.form_creation_in_progress = False

    try:

        while True:

            data = await websocket.receive_json()

            # 1. Chat message
            if data.get("type") == "message":

                try:
                    response = await run_in_threadpool(
                        chat_service.process_message,
                        data.get("message", "")
                    )

                    await websocket.send_json({
                        "type": "response",
                        "data": response
                    })

                except Exception as error:

                    await websocket.send_json({
                        "type": "error",
                        "message": str(error)
                    })

            # 2. Update edited drafts
            elif data.get("type") == "update_drafts":

                try:
                    raw_job_post = data.get("job_post", {})
                    raw_application_form = data.get(
                        "application_form", {}
                    )

                    job_post = prepare_job_post(raw_job_post)

                    job_post["job_description"] = (
                        raw_job_post.get("job_description", "")
                    )

                    application_form = prepare_application_form(
                        raw_application_form
                    )

                    original_questions = raw_application_form.get(
                        "questions", []
                    )

                    for prepared, original in zip(
                        application_form["questions"],
                        original_questions
                    ):
                        prepared["field_name"] = original.get("field_name")
                        prepared["validation"] = original.get(
                            "validation", {}
                        )

                    chat_service.current_job_post = job_post
                    chat_service.current_application_form = application_form

                    chat_service.job_post_approved = False
                    chat_service.application_form_approved = False

                    chat_service.created_post_id = None
                    chat_service.created_form_id = None
                    chat_service.created_form_url = None
                    chat_service.question_mapping = None

                    chat_service.recruitment_metadata_saved = False
                    chat_service.response_watch_created = False
                    chat_service.form_creation_in_progress = False

                    await websocket.send_json({
                        "type": "drafts_updated",
                        "message": "Both edited drafts saved successfully."
                    })

                except Exception as error:

                    await websocket.send_json({
                        "type": "error",
                        "message": str(error)
                    })

            # 3. Approve job post
            elif data.get("type") == "approve_job_post":

                if not chat_service.current_job_post:
                    await websocket.send_json({
                        "type": "error",
                        "message": "No job post is available."
                    })
                    continue

                chat_service.job_post_approved = True

                await websocket.send_json({
                    "type": "approval",
                    "job_post_approved": True,
                    "application_form_approved":
                        chat_service.application_form_approved
                })

                await create_form_if_approved()

            # 4. Approve application form
            elif data.get("type") == "approve_application_form":

                if not chat_service.current_application_form:
                    await websocket.send_json({
                        "type": "error",
                        "message": "No application form is available."
                    })
                    continue

                chat_service.application_form_approved = True

                await websocket.send_json({
                    "type": "approval",
                    "job_post_approved":
                        chat_service.job_post_approved,
                    "application_form_approved": True
                })

                await create_form_if_approved()

            # 5. Reset
            elif data.get("type") == "reset":

                active_sessions[session_id] = ChatService()
                chat_service = active_sessions[session_id]

                await websocket.send_json({
                    "type": "reset",
                    "message": "Conversation reset successfully."
                })

    except WebSocketDisconnect:
        print(f"HR disconnected: {session_id}")

    except Exception as error:
        print(f"WebSocket error for {session_id}: {error}")

# ============================================================
# EXTRACT JSON FROM MODEL RESPONSE
# ============================================================

def extract_json(response_text):

    if isinstance(response_text, list):
        response_text = "".join(
            item.get("text", "")
            if isinstance(item, dict)
            else str(item)
            for item in response_text
        )

    if not isinstance(response_text, str):
        raise ValueError(
            "Model returned an invalid response."
        )

    response_text = response_text.strip()

    response_text = re.sub(
        r"^```(?:json)?\s*",
        "",
        response_text,
        flags=re.IGNORECASE
    )

    response_text = re.sub(
        r"\s*```$",
        "",
        response_text
    )

    try:
        return json.loads(response_text)

    except json.JSONDecodeError:

        start = response_text.find("{")
        end = response_text.rfind("}")

        if start == -1 or end == -1:
            raise ValueError(
                "The AI did not return valid JSON."
            )

        try:
            return json.loads(
                response_text[start:end + 1]
            )

        except json.JSONDecodeError as error:
            raise ValueError(
                f"AI returned invalid JSON: {error}"
            )

def build_conversation_messages(conversation):

        SystemMessage(content=SYSTEM_PROMPT)

        content = item.get("content", "")

        if not isinstance(content, str):
            continue

        if role == "user":

            messages.append(
                HumanMessage(content=content)
            )

        elif role == "assistant":

            messages.append(
                AIMessage(content=content)
            )

    return messages

def validate_ai_response(result):

    if not isinstance(result, dict):
        raise ValueError(
            "AI response must be a JSON object."
        )

    if result.get("action") != "draft":
        raise ValueError(
            "AI did not return the expected draft action."
        )

    job_post = result.get("job_post")
    application_form = result.get("application_form")

    if not isinstance(job_post, dict):
        raise ValueError(
            "AI did not return a valid job post."
        )

    if not isinstance(application_form, dict):
        raise ValueError(
            "AI did not return a valid application form."
        )

    # Validate job description
    job_description = job_post.get("job_description")

    if (
        not isinstance(job_description, str)
        or not job_description.strip()
    ):
        raise ValueError(
            "AI response must contain a valid job_description."
        )

    # Validate form title
    form_title = application_form.get("form_title")

    if (
        not isinstance(form_title, str)
        or not form_title.strip()
    ):
        raise ValueError(
            "Application form title cannot be empty."
        )

    questions = application_form.get("questions")

    if not isinstance(questions, list) or not questions:
        raise ValueError(
            "Application form questions must be a non-empty list."
        )

    # Standard applicant fields
    required_fields = {
        "full_name",
        "email",
        "phone",
        "resume_file"
    }

    returned_fields = set()

    allowed_types = {
        "text",
        "email",
        "number",
        "yes_no",
        "multiple_choice",
        "dropdown",
        "date",
        "url",
        "file"
    }

    for question in questions:

        if not isinstance(question, dict):
            raise ValueError(
                "Invalid application form question."
            )

        required_keys = {
            "field_name",
            "question",
            "type",
            "required",
            "options",
            "validation"
        }

        missing_keys = required_keys - question.keys()

        if missing_keys:
            raise ValueError(
                "Application question is missing fields: "
                + ", ".join(sorted(missing_keys))
            )

        field_name = question["field_name"]
        question_text = question["question"]
        question_type = question["type"]

        if not isinstance(field_name, str) or not field_name.strip():
            raise ValueError(
                "Every question must have a valid field_name."
            )

        if field_name in returned_fields:
            raise ValueError(
                f"Duplicate application field: {field_name}"
            )

        returned_fields.add(field_name)

        if not isinstance(question_text, str) or not question_text.strip():
            raise ValueError(
                f"Invalid question text for {field_name}."
            )

        if question_type not in allowed_types:
            raise ValueError(
                f"Unsupported question type: {question_type}"
            )

        if not isinstance(question["required"], bool):
            raise ValueError(
                f"Invalid required value for {field_name}."
            )

        if not isinstance(question["options"], list):
            raise ValueError(
                f"Options must be a list for {field_name}."
            )

        if not isinstance(question["validation"], dict):
            raise ValueError(
                f"Validation must be an object for {field_name}."
            )

        if (
            question_type in {"multiple_choice", "dropdown"}
            and not question["options"]
        ):
            raise ValueError(
                f"{question_type} question {field_name} "
                "must contain options."
            )

        if (
            question_type not in {"multiple_choice", "dropdown"}
            and question["options"]
        ):
            raise ValueError(
                f"Question {field_name} must have options: []."
            )

    missing_fields = required_fields - returned_fields

    if missing_fields:
        raise ValueError(
            "Application form is missing standard questions: "
            + ", ".join(sorted(missing_fields))
        )

    return result

import json
from datetime import datetime

from backend.database.recruitment_db import get_recruitment_connection
from backend.services.google_form_service import get_forms_service


def _answer_value(answer):
    """Extract answer values from a Google Forms response."""
    text_answers = answer.get("textAnswers", {}).get("answers", [])

    values = [item.get("value", "") for item in text_answers]

    if not values:
        return None

    return values[0] if len(values) == 1 else values


def _get_form_metadata(form_id):
    connection = get_recruitment_connection()

    try:
        with connection.cursor() as cursor:
            cursor.execute(
                """
                SELECT post_id, question_mapping
                FROM recruitment_metadata
                WHERE form_id = %s
                """,
                (form_id,)
            )

            result = cursor.fetchone()

            if not result:
                raise ValueError(
                    f"No recruitment post found for form {form_id}"
                )

            post_id, question_mapping = result

            if isinstance(question_mapping, str):
                question_mapping = json.loads(question_mapping)

            return post_id, question_mapping or {}

    finally:
        connection.close()


def _save_applicant(form_id, response, post_id, question_mapping):
    response_id = response.get("responseId")

    if not response_id:
        raise ValueError("Google Forms response has no responseId")

    answers = response.get("answers", {})

    applicant = {
        "full_name": None,
        "email": None,
        "phone": None
    }

    additional_data = {}

    for question_id, answer in answers.items():
        field_name = question_mapping.get(question_id)

        if not field_name:
            continue

        value = _answer_value(answer)

        if field_name in applicant:
            applicant[field_name] = value
        elif field_name != "resume_file":
            additional_data[field_name] = value

    if not applicant["full_name"] or not applicant["email"]:
        raise ValueError(
            f"Required applicant details missing in response {response_id}. "
            "Check the question_mapping."
        )

    connection = get_recruitment_connection()

    try:
        with connection.cursor() as cursor:

            # Prevent duplicate applications.
            cursor.execute(
                """
                INSERT INTO recruitment_processed_responses
                    (form_id, response_id)
                VALUES (%s, %s)
                ON CONFLICT (form_id, response_id) DO NOTHING
                RETURNING response_id
                """,
                (form_id, response_id)
            )

            inserted = cursor.fetchone()

            if not inserted:
                connection.rollback()
                print(f"Duplicate response ignored: {response_id}")
                return False

            submitted_at = response.get("lastSubmittedTime")

            if submitted_at:
                submitted_at = datetime.fromisoformat(
                    submitted_at.replace("Z", "+00:00")
                )

            cursor.execute(
                """
                INSERT INTO recruitment_data (
                    post_id,
                    full_name,
                    email,
                    phone,
                    additional_data,
                    submitted_at
                )
                VALUES (%s, %s, %s, %s, %s, COALESCE(%s, CURRENT_TIMESTAMP))
                """,
                (
                    post_id,
                    applicant["full_name"],
                    applicant["email"],
                    applicant["phone"],
                    json.dumps(additional_data),
                    submitted_at
                )
            )

        connection.commit()

        print(
            f"Applicant saved: {applicant['email']} "
            f"for {post_id}"
        )

        return True

    except Exception:
        connection.rollback()
        raise

    finally:
        connection.close()


def process_form_responses(form_id):
    """Fetch and process all responses for a recruitment form."""

    post_id, question_mapping = _get_form_metadata(form_id)

    service = get_forms_service()

    page_token = None
    saved_count = 0

    while True:
        request_args = {
            "formId": form_id,
            "pageSize": 500
        }

        if page_token:
            request_args["pageToken"] = page_token

        result = service.forms().responses().list(
            **request_args
        ).execute()

        for response in result.get("responses", []):
            if _save_applicant(
                form_id,
                response,
                post_id,
                question_mapping
            ):
                saved_count += 1

        page_token = result.get("nextPageToken")

        if not page_token:
            break

    print(
        f"Response processing completed for {post_id}. "
        f"New applicants saved: {saved_count}"
    )

    return saved_count

def get_credentials():

    if not TOKEN_FILE.exists():
        raise FileNotFoundError(
            "Google authentication token not found. "
            "Please authenticate with Google first."
        )

    credentials = Credentials.from_authorized_user_file(
        str(TOKEN_FILE),
        scopes=FORMS_SCOPE
    )

    if credentials.expired and credentials.refresh_token:

        credentials.refresh(Request())

        TOKEN_FILE.write_text(
            credentials.to_json(),
            encoding="utf-8"
        )

    if not credentials.valid:
        raise ValueError(
            "Google credentials are invalid. "
            "Please authenticate with Google again."
        )

    return credentials

def separate_questions(questions):

    if not isinstance(questions, list) or not questions:
        raise ValueError(
            "Questions must be a non-empty list."
        )

    question_map = {}

    for index, question in enumerate(questions):

        if not isinstance(question, dict):
            raise ValueError(
                f"Question {index + 1} must be a dictionary."
            )

        field_name = question.get("field_name")

        if not isinstance(field_name, str) or not field_name.strip():
            raise ValueError(
                f"Question {index + 1} is missing field_name."
            )

        field_name = field_name.strip()

        if field_name in question_map:
            raise ValueError(
                f"Duplicate field_name: {field_name}"
            )

        question_map[field_name] = question

    missing_fields = [
        field
        for field in STANDARD_FIELD_ORDER
        if field not in question_map
    ]

    if missing_fields:
        raise ValueError(
            f"Missing standard applicant fields: {missing_fields}"
        )

    standard_questions = []
    additional_questions = []

    for field_name in STANDARD_FIELD_ORDER:

        question = question_map[field_name]

        actual_type = (
            question.get("type", "")
            .lower()
            .strip()
            .replace("-", "_")
        )

        expected_type = STANDARD_FIELDS[field_name]

        if actual_type != expected_type:
            raise ValueError(
                f"Invalid type for {field_name}. "
                f"Expected '{expected_type}', got '{actual_type}'."
            )

        standard_questions.append(question)

    for field_name, question in question_map.items():

        if field_name not in STANDARD_FIELDS:
            additional_questions.append(question)

    return standard_questions, additional_questions

def prepare_google_form_questions(questions):

    if not isinstance(questions, list) or not questions:
        raise ValueError(
            "At least one application question is required."
        )

    requests = []
    skipped_questions = []
    question_field_names = []

    for index, question in enumerate(questions):

        question_text = question.get("question", "").strip()

        question_type = (
            question.get("type", "")
            .lower()
            .strip()
            .replace("-", "_")
        )

        field_name = question.get("field_name")

        required = question.get("required", False)
        options = question.get("options", [])

        if not question_text:
            raise ValueError(
                f"Question {index + 1} has no question text."
            )

        if not isinstance(field_name, str) or not field_name.strip():
            raise ValueError(
                f"Question {index + 1} has no field_name."
            )

        if not isinstance(required, bool):
            raise ValueError(
                f"Invalid required value for question {index + 1}."
            )

        if not isinstance(options, list):
            raise ValueError(
                f"Options must be a list for question {index + 1}."
            )

        # FILE UPLOAD

        if question_type in ("file", "file_upload"):

            skipped_questions.append({
                "field_name": field_name,
                "question": question_text,
                "reason": "Handled separately through FastAPI."
            })

            continue

        # SHORT ANSWER

        if question_type in (
            "text",
            "short_answer",
            "email",
            "number",
            "url"
        ):

            question_body = {
                "textQuestion": {
                    "paragraph": False
                }
            }

        # PARAGRAPH

        elif question_type == "paragraph":

            question_body = {
                "textQuestion": {
                    "paragraph": True
                }
            }

        # YES / NO

        elif question_type == "yes_no":

            question_body = {
                "choiceQuestion": {
                    "type": "RADIO",
                    "options": [
                        {"value": "Yes"},
                        {"value": "No"}
                    ],
                    "shuffle": False
                }
            }

        # MULTIPLE CHOICE

        elif question_type == "multiple_choice":

            if not options:
                raise ValueError(
                    f"Multiple-choice question {index + 1} "
                    "must have options."
                )

            question_body = {
                "choiceQuestion": {
                    "type": "RADIO",
                    "options": [
                        {"value": str(option)}
                        for option in options
                    ],
                    "shuffle": False
                }
            }

        # CHECKBOX

        elif question_type == "checkbox":

            if not options:
                raise ValueError(
                    f"Checkbox question {index + 1} "
                    "must have options."
                )

            question_body = {
                "choiceQuestion": {
                    "type": "CHECKBOX",
                    "options": [
                        {"value": str(option)}
                        for option in options
                    ],
                    "shuffle": False
                }
            }

        # DROPDOWN

        elif question_type == "dropdown":

            if not options:
                raise ValueError(
                    f"Dropdown question {index + 1} "
                    "must have options."
                )

            question_body = {
                "choiceQuestion": {
                    "type": "DROP_DOWN",
                    "options": [
                        {"value": str(option)}
                        for option in options
                    ],
                    "shuffle": False
                }
            }

        # DATE

        elif question_type == "date":

            question_body = {
                "dateQuestion": {
                    "includeTime": False,
                    "includeYear": True
                }
            }

        else:

            raise ValueError(
                f"Unsupported question type '{question_type}' "
                f"for question {index + 1}."
            )

        item = {
            "title": question_text,
            "questionItem": {
                "question": {
                    "required": required,
                    **question_body
                }
            }
        }

        requests.append({
            "createItem": {
                "item": item,
                "location": {
                    "index": len(requests)
                }
            }
        })

        question_field_names.append(field_name)

    if not requests:
        raise ValueError(
            "No supported Google Form questions found."
        )

    return requests, skipped_questions, question_field_names

# CREATE GOOGLE FORM

def create_google_form(title, questions, job_description):

    if not isinstance(title, str) or not title.strip():
        raise ValueError(
            "A valid Google Form title is required."
        )

    if (
        not isinstance(job_description, str)
        or not job_description.strip()
    ):
        raise ValueError(
            "A valid job description is required."
        )

    # Validate questions

    standard_questions, additional_questions = separate_questions(
        questions
    )

    ordered_questions = (
        standard_questions + additional_questions
    )

    # Prepare Google API requests

    requests, skipped_questions, question_field_names = (
        prepare_google_form_questions(ordered_questions)
    )

    service = get_forms_service()

    # --------------------------------------------------------
    # 1. CREATE GOOGLE FORM
    # --------------------------------------------------------

    form_response = service.forms().create(
        body={
            "info": {
                "title": title.strip()
            }
        }
    ).execute()

    form_id = form_response.get("formId")

    if not form_id:
        raise RuntimeError(
            "Google Form creation did not return a form ID."
        )

    # --------------------------------------------------------
    # 2. RENAME GOOGLE DRIVE FILE
    # --------------------------------------------------------

    drive_file_name = rename_form_in_drive(
        form_id,
        title
    )

    # --------------------------------------------------------
    # 3. ADD DESCRIPTION AND QUESTIONS
    # --------------------------------------------------------

    batch_requests = [
        {
            "updateFormInfo": {
                "info": {
                    "description": job_description.strip()
                },
                "updateMask": "description"
            }
        }
    ]

    batch_requests.extend(requests)

    batch_response = service.forms().batchUpdate(
        formId=form_id,
        body={
            "requests": batch_requests
        }
    ).execute()

    replies = batch_response.get("replies", [])

    if len(replies) != len(requests) + 1:
        raise RuntimeError(
            "Google Forms returned an unexpected number of replies."
        )

    # --------------------------------------------------------
    # 4. CREATE QUESTION ID TO FIELD NAME MAPPING
    # --------------------------------------------------------

    question_mapping = {}

    for index, field_name in enumerate(
        question_field_names,
        start=1
    ):

        create_item_response = (
            replies[index].get("createItem", {})
        )

        question_ids = create_item_response.get(
            "questionId",
            []
        )

        if not question_ids:
            raise RuntimeError(
                f"Google did not return a question ID for {field_name}."
            )

        question_mapping[question_ids[0]] = field_name

    # --------------------------------------------------------
    # 5. RETRIEVE COMPLETED FORM
    # --------------------------------------------------------

    form_details = service.forms().get(
        formId=form_id
    ).execute()

    return {
        "form_id": form_id,
        "title": form_details.get("info", {}).get("title"),
        "description": form_details.get("info", {}).get("description"),
        "form_url": form_details.get("responderUri"),
        "drive_file_name": drive_file_name,
        "skipped_questions": skipped_questions,
        "question_mapping": question_mapping
    }

JOB_POST_FIELDS = [
    "job_title",
    "department",
    "location",
    "work_mode",
    "employment_type",
    "experience",
    "salary_range",
    "role_summary",
    "responsibilities",
    "required_skills",
    "preferred_skills",
    "education",
    "benefits",
    "application_instructions"
]

LIST_FIELDS = [
    "responsibilities",
    "required_skills",
    "preferred_skills",
    "education",
    "benefits"
]


def prepare_job_post(job_post):

    if not isinstance(job_post, dict):
        raise ValueError("Job post must be a JSON object.")

    prepared_job_post = {}

    for field in JOB_POST_FIELDS:

        value = job_post.get(
            field,
            [] if field in LIST_FIELDS else ""
        )

        if field in LIST_FIELDS:

            if value is None:
                value = []

            if not isinstance(value, list):
                value = [value]

            prepared_job_post[field] = value

        else:

            if value is None:
                value = ""

            prepared_job_post[field] = value

    return prepared_job_post


def prepare_application_form(application_form):

    if not isinstance(application_form, dict):
        raise ValueError("Application form must be a JSON object.")

    form_title = application_form.get("form_title", "")

    if form_title is None:
        form_title = ""

    questions = application_form.get("questions", [])

    if questions is None:
        questions = []

    if not isinstance(questions, list):
        raise ValueError("Application questions must be a list.")

    prepared_questions = []

    for index, question in enumerate(questions, start=1):

        if not isinstance(question, dict):
            raise ValueError(
                f"Question {index} must be a JSON object."
            )

        question_text = question.get("question", "")
        question_type = question.get("type", "")
        required = question.get("required", False)
        options = question.get("options", [])

        if question_text is None:
            question_text = ""

        if question_type is None:
            question_type = ""

        if not isinstance(question_text, str):
            raise ValueError(
                f"Invalid question text for question {index}."
            )

        if not isinstance(question_type, str):
            raise ValueError(
                f"Invalid question type for question {index}."
            )

        if not isinstance(required, bool):
            raise ValueError(
                f"Required must be true or false for question {index}."
            )

        if options is None:
            options = []

        if not isinstance(options, list):
            raise ValueError(
                f"Options must be a list for question {index}."
            )

        prepared_questions.append({
            "question": question_text,
            "type": question_type,
            "required": required,
            "options": options
        })

    return {
        "form_title": form_title,
        "questions": prepared_questions
    }

import json
import threading
import time

from google.cloud import pubsub_v1

from backend.services.applicant_response_service import (
    process_form_responses
)

PROJECT_ID = "manifest-life-513710-d4"
SUBSCRIPTION_ID = "nexora-form-responses-sub"

SUBSCRIPTION_PATH = (
    f"projects/{PROJECT_ID}/subscriptions/{SUBSCRIPTION_ID}"
)

subscriber = pubsub_v1.SubscriberClient()

_listener_thread = None
_stop_event = threading.Event()


def _process_message(received_message):
    message = received_message.message

    try:
        # Google Forms sends notification details as Pub/Sub attributes.
        attributes = dict(message.attributes)

        form_id = attributes.get("formId")
        event_type = attributes.get("eventType")

        if not form_id:
            raise ValueError(
                f"Notification does not contain formId: {attributes}"
            )

        if event_type != "RESPONSES":
            print(f"Ignoring notification event: {event_type}")
            return True

        print(f"Google Forms response notification received: {form_id}")

        saved_count = process_form_responses(form_id)

        print(
            f"Processing completed. New applicants saved: {saved_count}"
        )

        return True

    except Exception as error:
        print(f"Error processing notification: {error}")
        print(f"Message attributes: {dict(message.attributes)}")
        return False


def _listen():
    print("Nexora Pub/Sub listener started.")

    while not _stop_event.is_set():
        try:
            response = subscriber.pull(
                request={
                    "subscription": SUBSCRIPTION_PATH,
                    "max_messages": 10
                },
                retry=None,
                timeout=60
            )

            for received_message in response.received_messages:
                if _stop_event.is_set():
                    break

                success = _process_message(received_message)

                if success:
                    subscriber.acknowledge(
                        request={
                            "subscription": SUBSCRIPTION_PATH,
                            "ack_ids": [received_message.ack_id]
                        }
                    )
                else:
                    # Release the message for redelivery.
                    subscriber.modify_ack_deadline(
                        request={
                            "subscription": SUBSCRIPTION_PATH,
                            "ack_ids": [received_message.ack_id],
                            "ack_deadline_seconds": 0
                        }
                    )

        except Exception as error:
            if not _stop_event.is_set():
                print(f"Pub/Sub listener error: {error}")
                time.sleep(5)

    print("Nexora Pub/Sub listener stopped.")


def start_pubsub_listener():
    global _listener_thread

    if _listener_thread and _listener_thread.is_alive():
        print("Pub/Sub listener is already running.")
        return

    _stop_event.clear()

    _listener_thread = threading.Thread(
        target=_listen,
        daemon=True
    )

    _listener_thread.start()


def stop_pubsub_listener():
    _stop_event.set()

    if _listener_thread and _listener_thread.is_alive():
        _listener_thread.join(timeout=5)

    subscriber.close()
